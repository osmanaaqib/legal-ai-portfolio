        "# MENA Jurisdiction Router\n",

        "#A confidence-aware retrieval system for UAE legal text that routes queries across\n",
        "#Federal, DIFC, and ADGM regimes without silently guessing when a query is ambiguous.\n",

        "##The UAE has three overlapping legal systems: onshore Federal law, the DIFC\n",

        "#A legally correct DIFC answer can be wrong for an onshore dispute. Generic\n",

        "#most semantically similar text regardless of jurisdiction, silently mixing\n",
        "#regimes. In legal work, a correct answer from the wrong jurisdiction is worse\n",
        "#than no answer.\n",

        "# The core of this project. The router classifies a query by jurisdiction\n",
        "# using keyword markers, then returns a confidence score and a filter.\n",
        "#\n",
        "# Design principles:\n",
        "#   1. DIFC and ADGM are specific regimes. They are matched first and\n",
        "#      independently. Federal is the default — returned only when no\n",
        "#      free-zone marker is present.\n",
        "#   2. Markers are normalized once at definition time so Arabic markers\n",
        "#      containing أ (e.g., \"أبوظبي\") match normalized query text.\n",
        "#   3. When DIFC and ADGM markers tie, the query is genuinely ambiguous\n",
        "#      about jurisdiction. The router flags this as low confidence and\n",
        "#      returns filter=None so retrieval runs across all regimes rather\n",
        "#      than silently guessing. This is the correct behaviour for legal AI.\n",
        "\n",
        "def _norm(s: str) -> str:\n",
        "    return normalize_arabic(s).lower()\n",
        "\n",
        "# Pre-normalize markers. If I don't do this, Arabic markers with أ won't\n",
        "# match Arabic queries with ا after normalization.\n",
        "DIFC_MARKERS = [_norm(m) for m in [\n",
        "    \"difc\",\n",
