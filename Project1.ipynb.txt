        "# then ask a second LLM (the \"judge\") to score the output against a rubric.\n",
        "# The judge sees the article text as the source of truth, so it's checking\n",

        "#\n",
        "# Three things about gpt-oss-120b I had to work around:\n",
        "# 1. It's a reasoning model, and it leaks reasoning preamble into the\n",
        "#    output even in JSON mode. So I extract JSON with a depth-tracking\n",

        "# 2. Groq requires the literal word \"json\" to appear in the prompt when\n",

        "#    cap at 500 to stay under the ceiling.\n",
        "\n",
        "def extract_json(text):\n",
        "    \"\"\"Pull the first valid JSON object out of a string, ignoring preamble\n",
        "    and markdown fences. gpt-oss-120b tends to leak both.\"\"\"\n",
        "    if not text:\n",
        "        return None\n",
        "    text = re.sub(r'```(?:json)?\\s*', '', text)\n",
        "    text = text.replace('```', '')\n",
        "    start = text.find('{')\n",
        "    if start == -1:\n",
        "        return None\n",
        "    depth = 0\n",
        "    for i in range(start, len(text)):\n",
        "        if text[i] == '{':\n",
        "            depth += 1\n",
        "        elif text[i] == '}':\n",
        "            depth -= 1\n",
        "            if depth == 0:\n",
        "                return text[start:i+1]\n",
